## Intent

Define the skeleton of an algorithm in a method, deferring some steps to subclasses. Template Method lets subclasses redefine certain steps without changing the algorithm's structure.

## Data Export Example

```csharp
public abstract class DataExporter
{
    // Template method - defines algorithm skeleton
    public void Export(IEnumerable<Data> data)
    {
        var processed = Transform(data);
        var formatted = Format(processed);
        Save(formatted);
        NotifyCompletion();
    }

    // Common implementation
    protected virtual void NotifyCompletion() 
        => Console.WriteLine("Export completed");

    // Steps to be implemented by subclasses
    protected abstract IEnumerable<Data> Transform(IEnumerable<Data> data);
    protected abstract string Format(IEnumerable<Data> data);
    protected abstract void Save(string content);
}

public class CsvExporter : DataExporter
{
    protected override IEnumerable<Data> Transform(IEnumerable<Data> data)
        => data.Where(d => d.IsValid);

    protected override string Format(IEnumerable<Data> data)
        => string.Join("\n", data.Select(d => $"{d.Id},{d.Name},{d.Value}"));

    protected override void Save(string content)
        => File.WriteAllText("export.csv", content);
}

public class JsonExporter : DataExporter
{
    protected override IEnumerable<Data> Transform(IEnumerable<Data> data)
        => data;

    protected override string Format(IEnumerable<Data> data)
        => JsonSerializer.Serialize(data);

    protected override void Save(string content)
        => File.WriteAllText("export.json", content);
}
```

## Hook Methods

```csharp
public abstract class DataExporter
{
    public void Export(IEnumerable<Data> data)
    {
        if (ShouldValidate())  // Hook - optional override
            Validate(data);
        
        // ... rest of algorithm
    }

    // Hook with default implementation
    protected virtual bool ShouldValidate() => true;
    protected virtual void Validate(IEnumerable<Data> data) { }
}
```
