## Intent

Define a family of algorithms, encapsulate each one, and make them interchangeable. Strategy lets the algorithm vary independently from clients that use it.

## Sorting Example

```csharp
public interface ISortStrategy<T>
{
    IEnumerable<T> Sort(IEnumerable<T> items);
}

public class QuickSort<T> : ISortStrategy<T> where T : IComparable<T>
{
    public IEnumerable<T> Sort(IEnumerable<T> items) => items.OrderBy(x => x);
}

public class BubbleSort<T> : ISortStrategy<T> where T : IComparable<T>
{
    public IEnumerable<T> Sort(IEnumerable<T> items)
    {
        // Bubble sort implementation
    }
}

public class Sorter<T>
{
    private ISortStrategy<T> _strategy;

    public Sorter(ISortStrategy<T> strategy) => _strategy = strategy;

    public void SetStrategy(ISortStrategy<T> strategy) => _strategy = strategy;

    public IEnumerable<T> Sort(IEnumerable<T> items) => _strategy.Sort(items);
}
```

## Payment Strategies

```csharp
public interface IPaymentStrategy
{
    Task<PaymentResult> PayAsync(decimal amount);
}

public class CreditCardPayment : IPaymentStrategy
{
    private readonly string _cardNumber;
    public CreditCardPayment(string cardNumber) => _cardNumber = cardNumber;
    public async Task<PaymentResult> PayAsync(decimal amount) { /* ... */ }
}

public class PayPalPayment : IPaymentStrategy
{
    private readonly string _email;
    public PayPalPayment(string email) => _email = email;
    public async Task<PaymentResult> PayAsync(decimal amount) { /* ... */ }
}

public class ShoppingCart
{
    private readonly List<Item> _items = new();

    public async Task<PaymentResult> CheckoutAsync(IPaymentStrategy paymentStrategy)
    {
        var total = _items.Sum(i => i.Price);
        return await paymentStrategy.PayAsync(total);
    }
}

// Usage
var cart = new ShoppingCart();
await cart.CheckoutAsync(new CreditCardPayment("4111111111111111"));
await cart.CheckoutAsync(new PayPalPayment("user@example.com"));
```
