## Intent

Provide a surrogate or placeholder for another object to control access to it.

## Types of Proxies

- **Virtual Proxy:** Lazy initialization of expensive objects
- **Protection Proxy:** Access control
- **Caching Proxy:** Cache results
- **Logging Proxy:** Add logging

## Caching Proxy

```csharp
public interface IWeatherService
{
    Task<Weather> GetWeatherAsync(string city);
}

public class WeatherService : IWeatherService
{
    public async Task<Weather> GetWeatherAsync(string city)
    {
        // Expensive API call
        await Task.Delay(1000);
        return new Weather { City = city, Temperature = 22 };
    }
}

public class CachingWeatherProxy : IWeatherService
{
    private readonly IWeatherService _service;
    private readonly Dictionary<string, (Weather Data, DateTime Expiry)> _cache = new();

    public CachingWeatherProxy(IWeatherService service) => _service = service;

    public async Task<Weather> GetWeatherAsync(string city)
    {
        if (_cache.TryGetValue(city, out var cached) && cached.Expiry > DateTime.Now)
            return cached.Data;

        var weather = await _service.GetWeatherAsync(city);
        _cache[city] = (weather, DateTime.Now.AddMinutes(10));
        return weather;
    }
}
```

## Protection Proxy

```csharp
public class SecureDocumentProxy : IDocument
{
    private readonly IDocument _document;
    private readonly IUserContext _userContext;

    public SecureDocumentProxy(IDocument document, IUserContext userContext)
    {
        _document = document;
        _userContext = userContext;
    }

    public void Read() => _document.Read();  // Anyone can read

    public void Write(string content)
    {
        if (!_userContext.HasPermission("document:write"))
            throw new UnauthorizedAccessException();
        
        _document.Write(content);
    }

    public void Delete()
    {
        if (!_userContext.IsAdmin)
            throw new UnauthorizedAccessException();
        
        _document.Delete();
    }
}
```
