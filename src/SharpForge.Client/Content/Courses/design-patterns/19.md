## Intent

Encapsulate a request as an object, allowing you to parameterize clients with different requests, queue or log requests, and support undoable operations.

## Text Editor Example

```csharp
public interface ICommand
{
    void Execute();
    void Undo();
}

public class TextEditor
{
    public string Text { get; set; } = "";
}

public class AddTextCommand : ICommand
{
    private readonly TextEditor _editor;
    private readonly string _text;

    public AddTextCommand(TextEditor editor, string text)
    {
        _editor = editor;
        _text = text;
    }

    public void Execute() => _editor.Text += _text;
    public void Undo() => _editor.Text = _editor.Text[..^_text.Length];
}

public class CommandHistory
{
    private readonly Stack<ICommand> _history = new();

    public void ExecuteCommand(ICommand command)
    {
        command.Execute();
        _history.Push(command);
    }

    public void Undo()
    {
        if (_history.Count > 0)
            _history.Pop().Undo();
    }
}
```

## MediatR (CQRS)

```csharp
// Command
public record CreateOrderCommand(string CustomerId, List<OrderItem> Items) 
    : IRequest<OrderResult>;

// Handler
public class CreateOrderHandler : IRequestHandler<CreateOrderCommand, OrderResult>
{
    public async Task<OrderResult> Handle(CreateOrderCommand request, CancellationToken ct)
    {
        // Process command
    }
}

// Usage
var result = await _mediator.Send(new CreateOrderCommand(customerId, items));
```
