## Intent

Attach additional responsibilities to an object dynamically. Decorators provide a flexible alternative to subclassing for extending functionality.

## Coffee Shop Example

```csharp
public interface ICoffee
{
    string Description { get; }
    decimal Cost { get; }
}

public class Espresso : ICoffee
{
    public string Description => "Espresso";
    public decimal Cost => 2.00m;
}

// Base decorator
public abstract class CoffeeDecorator : ICoffee
{
    protected readonly ICoffee _coffee;
    protected CoffeeDecorator(ICoffee coffee) => _coffee = coffee;
    public virtual string Description => _coffee.Description;
    public virtual decimal Cost => _coffee.Cost;
}

public class Milk : CoffeeDecorator
{
    public Milk(ICoffee coffee) : base(coffee) { }
    public override string Description => $"{_coffee.Description}, Milk";
    public override decimal Cost => _coffee.Cost + 0.50m;
}

public class Caramel : CoffeeDecorator
{
    public Caramel(ICoffee coffee) : base(coffee) { }
    public override string Description => $"{_coffee.Description}, Caramel";
    public override decimal Cost => _coffee.Cost + 0.75m;
}

// Usage - stack decorators!
ICoffee order = new Espresso();
order = new Milk(order);
order = new Caramel(order);
Console.WriteLine($"{order.Description}: ${order.Cost}");
// Output: Espresso, Milk, Caramel: $3.25
```

## Real-World: Logging Decorator

```csharp
public interface IOrderService
{
    Task<Order> CreateOrderAsync(OrderRequest request);
}

public class OrderService : IOrderService
{
    public async Task<Order> CreateOrderAsync(OrderRequest request)
    {
        // Actual implementation
    }
}

public class LoggingOrderService : IOrderService
{
    private readonly IOrderService _inner;
    private readonly ILogger _logger;

    public LoggingOrderService(IOrderService inner, ILogger logger)
    {
        _inner = inner;
        _logger = logger;
    }

    public async Task<Order> CreateOrderAsync(OrderRequest request)
    {
        _logger.LogInformation("Creating order...");
        var result = await _inner.CreateOrderAsync(request);
        _logger.LogInformation($"Order {result.Id} created");
        return result;
    }
}
```
