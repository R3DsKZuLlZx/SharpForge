## Intent

Compose objects into tree structures to represent part-whole hierarchies. Composite lets clients treat individual objects and compositions uniformly.

## File System Example

```csharp
public interface IFileSystemItem
{
    string Name { get; }
    long GetSize();
    void Display(int indent = 0);
}

// Leaf
public class File : IFileSystemItem
{
    public string Name { get; }
    public long Size { get; }

    public File(string name, long size) => (Name, Size) = (name, size);
    public long GetSize() => Size;
    public void Display(int indent = 0) 
        => Console.WriteLine($"{new string(' ', indent)}- {Name} ({Size} bytes)");
}

// Composite
public class Folder : IFileSystemItem
{
    public string Name { get; }
    private readonly List<IFileSystemItem> _items = new();

    public Folder(string name) => Name = name;

    public void Add(IFileSystemItem item) => _items.Add(item);
    public void Remove(IFileSystemItem item) => _items.Remove(item);

    public long GetSize() => _items.Sum(i => i.GetSize());

    public void Display(int indent = 0)
    {
        Console.WriteLine($"{new string(' ', indent)}+ {Name}/");
        foreach (var item in _items)
            item.Display(indent + 2);
    }
}
```

## Usage

```csharp
var root = new Folder("root");
var documents = new Folder("Documents");
var photos = new Folder("Photos");

documents.Add(new File("resume.pdf", 50000));
documents.Add(new File("notes.txt", 1000));
photos.Add(new File("vacation.jpg", 2000000));

root.Add(documents);
root.Add(photos);
root.Add(new File("readme.md", 500));

root.Display();
Console.WriteLine($"Total size: {root.GetSize()} bytes");
```
