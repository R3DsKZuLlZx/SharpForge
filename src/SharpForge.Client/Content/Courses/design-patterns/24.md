## Real-World Example: E-Commerce Order System

Let's combine multiple patterns in a realistic scenario.

## Patterns Used

```csharp
// FACTORY - Create payment processors
public class PaymentProcessorFactory
{
    public IPaymentProcessor Create(PaymentMethod method) => method switch
    {
        PaymentMethod.CreditCard => new CreditCardProcessor(),
        PaymentMethod.PayPal => new PayPalProcessor(),
        _ => throw new ArgumentException()
    };
}

// STRATEGY - Discount calculation
public interface IDiscountStrategy
{
    decimal Calculate(Order order);
}

// DECORATOR - Add logging to services
public class LoggingOrderService : IOrderService
{
    private readonly IOrderService _inner;
    private readonly ILogger _logger;
    
    public async Task<Order> CreateAsync(OrderRequest request)
    {
        _logger.LogInformation("Creating order...");
        var order = await _inner.CreateAsync(request);
        _logger.LogInformation($"Order {order.Id} created");
        return order;
    }
}

// OBSERVER - Notify on order events
public class OrderCreatedEvent
{
    public Guid OrderId { get; init; }
    public string CustomerEmail { get; init; }
}

public class EmailNotificationHandler : IEventHandler<OrderCreatedEvent>
{
    public Task HandleAsync(OrderCreatedEvent e) => 
        _emailService.SendOrderConfirmation(e.CustomerEmail, e.OrderId);
}
```

## Combined in a Service

```csharp
public class OrderService : IOrderService
{
    private readonly IOrderRepository _repository;
    private readonly IPaymentProcessorFactory _paymentFactory;
    private readonly IDiscountStrategy _discountStrategy;
    private readonly IEventPublisher _events;

    public async Task<OrderResult> CreateOrderAsync(CreateOrderRequest request)
    {
        // Create order (Builder pattern internally)
        var order = Order.Create(request.CustomerId);
        foreach (var item in request.Items)
            order.AddItem(item);

        // Apply discount (Strategy)
        var discount = _discountStrategy.Calculate(order);
        order.ApplyDiscount(discount);

        // Process payment (Factory)
        var processor = _paymentFactory.Create(request.PaymentMethod);
        var paymentResult = await processor.ProcessAsync(order.Total);
        if (!paymentResult.Success)
            return OrderResult.Failed(paymentResult.Error);

        // Save order (Repository)
        await _repository.AddAsync(order);

        // Publish event (Observer)
        await _events.PublishAsync(new OrderCreatedEvent 
        { 
            OrderId = order.Id, 
            CustomerEmail = request.Email 
        });

        return OrderResult.Success(order);
    }
}
```
